\[
\chi_{B,D}(A')=-\chi_{B,D}(A),
\qquad
\chi_{B,D}(X)=\frac{X-B}{X-D},
\]

\[
A'=\frac{(A-B)D+(A-D)B}{2A-B-D}.
\]
