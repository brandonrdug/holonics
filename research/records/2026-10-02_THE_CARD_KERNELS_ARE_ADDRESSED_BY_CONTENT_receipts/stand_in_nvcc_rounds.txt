The stand-in nvcc:
#!/bin/bash
# A stand-in nvcc for testing the build script's content address: it logs each call and writes a
# nonempty image to the path after -o.
log="$(dirname "$0")/calls.log"
if [ "$1" = "--version" ]; then echo "fake nvcc, release ${FAKE_NVCC_RELEASE:-1}"; exit 0; fi
echo "$*" >> "$log"
out=""; prev=""
for a in "$@"; do if [ "$prev" = "-o" ]; then out="$a"; fi; prev="$a"; done
printf '// fake image for %s\n' "$*" > "$out"
