def classifyDemand (x : CompositionDemand) : ℕ :=
  if x.predecessor && x.forwardAvailable && !x.reverseAvailable then 0
  else if x.forwardAvailable && x.reverseAvailable && decide (x.forward = x.reverse) then 1
  else if x.contact && x.forwardAvailable && x.reverseAvailable then 2
  else if decide (x.forward.obstruction ≠ 0) && !x.forwardAvailable then 3 else 4

  {a := x.a*y.a+x.b*y.c, b := x.a*y.b+x.b*y.d,
   c := x.c*y.a+x.d*y.c, d := x.c*y.b+x.d*y.d}

  {a := x.a-y.a, b := x.b-y.b, c := x.c-y.c, d := x.d-y.d}
