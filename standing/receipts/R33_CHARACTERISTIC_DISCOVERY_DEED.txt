structure Matrix2 where
  a : ℤ
  b : ℤ
  c : ℤ
  d : ℤ
  deriving DecidableEq, Repr

def matrixMultiply (u v : Matrix2) : Matrix2 :=
  {a:=u.a*v.a+u.b*v.c,b:=u.a*v.b+u.b*v.d,c:=u.c*v.a+u.d*v.c,d:=u.c*v.b+u.d*v.d}
def matrixInverse (u : Matrix2) : Matrix2 := {a:=u.d,b:=-u.b,c:=-u.c,d:=u.a}
def matrixTrace (u : Matrix2) : ℤ := u.a+u.d
